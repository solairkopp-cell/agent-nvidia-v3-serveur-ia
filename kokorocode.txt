from __future__ import annotations

import asyncio
import logging

import numpy as np
import onnxruntime as ort
import soxr
from kokoro_onnx import Kokoro

import config


class PiperTTSService:

    def __init__(self):
        self._lock = asyncio.Lock()
        self._kokoro: Kokoro | None = None
        self._sample_rate = 24000
        self._voice = "af_bella"

    async def startup(self):
        logger = logging.getLogger(__name__)

        def _load():
            providers = ["CPUExecutionProvider"]
            session = ort.InferenceSession(config.KOKORO_MODEL_PATH, providers=providers)
            return Kokoro.from_session(session, config.KOKORO_VOICES_PATH)

        self._kokoro = await asyncio.to_thread(_load)
        logger.info("Kokoro ONNX GPU ready voice=%s", self._voice)

    async def shutdown(self):
        self._kokoro = None

    async def synthesize(self, text: str) -> tuple[np.ndarray, int]:
        target_rate = 48000
        text = (text or "").strip()
        if not text:
            return np.array([], dtype=np.float32), target_rate

        async with self._lock:
            def _do():
                samples, rate = self._kokoro.create(
                    text,
                    voice=self._voice,
                    speed=1.0,
                    lang="en-us",
                )
                samples_48k = soxr.resample(samples.astype(np.float32), rate, target_rate)
                return samples_48k.astype(np.float32), target_rate

            return await asyncio.to_thread(_do)

    def apply_fade_out(self, samples: np.ndarray, rate: int, duration_ms: int = 150) -> np.ndarray:
        if not isinstance(samples, np.ndarray):
            samples = np.asarray(samples, dtype=np.float32)
        if samples.size == 0:
            return samples
        n = max(1, min(int(rate * duration_ms / 1000.0), samples.shape[0]))
        fade = np.exp(np.linspace(0.0, -6.0, num=n, dtype=np.float32))
        out = samples.copy()
        out[-n:] *= fade
        return out

    async def health_check(self) -> bool:
        try:
            samples, rate = await self.synthesize("test.")
            return isinstance(samples, np.ndarray) and samples.size > 0
        except Exception:
            return False
